3. MODELO: STACK MONOTÔNICA (Próximo Maior Elemento / Next Greater Element)
-------------------------------------------------------------------------------
// Retorna um vetor onde ans[i] contém o ÍNDICE do próximo elemento maior que v[i].
// Se não houver elemento maior à direita, ans[i] permanece -1.
// Complexidade de Tempo: O(N) | Espaço: O(N)

vector<int> next_greater_element(const vector<int>& v) {
    int n = v.size();
    vector<int> ans(n, -1);
    stack<int> st;

    for (int i = 0; i < n; i++) {
        while (!st.empty() && v[st.top()] < v[i]) {
            int idx = st.top();
            st.pop();
            ans[idx] = i;
        }
        st.push(i);
    }

    return ans;
}
