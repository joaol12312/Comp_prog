-------------------------------------------------------------------------------
1. STACK (Pilha - LIFO: Last-In, First-Out)
-------------------------------------------------------------------------------
stack<int> st;          // Criação da pilha: O(1)
st.push(5);             // Adiciona o elemento 5 ao topo da pilha: O(1)
st.top();               // Retorna o elemento do topo da pilha: O(1)
st.pop();               // Remove o elemento do topo da pilha: O(1)
st.empty();             // Retorna true se a pilha estiver vazia: O(1)
st.size();              // Retorna a quantidade de elementos: O(1)


-------------------------------------------------------------------------------
2. QUEUE (Fila - FIFO: First-In, First-Out)
-------------------------------------------------------------------------------
queue<int> q;           // Criação da fila: O(1)
q.push(5);              // Adiciona o elemento 5 ao final da fila: O(1)
q.front();              // Retorna o elemento do início da fila: O(1)
q.pop();                // Remove o elemento do início da fila: O(1)
q.empty();              // Retorna true se a fila estiver vazia: O(1)
q.size();               // Retorna a quantidade de elementos: O(1)


-------------------------------------------------------------------------------
3. MODELO: STACK MONOTÔNICA (Próximo Maior Elemento / Next Greater Element)
-------------------------------------------------------------------------------
// Retorna um vetor onde ans[i] contém o ÍNDICE do próximo elemento maior que v[i].
// Se não houver elemento maior à direita, ans[i] permanece -1.
// Complexidade de Tempo: O(N) | Espaço: O(N)

vector<int> next_greater_element(const vector<int>& v) {
    int n = v.size();
    vector<int> ans(n, -1);
    stack<int> st;

    for (int i = 0; i < n; i++) {
        while (!st.empty() && v[st.top()] < v[i]) {
            int idx = st.top();
            st.pop();
            ans[idx] = i;
        }
        st.push(i);
    }

    return ans;
}
