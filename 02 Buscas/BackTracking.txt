void backtrack(int estado, int tamanho_alvo, vector<int>& solucao_parcial) {
    // 1. CASO BASE (Condição de Parada / Sucesso)
    if (estado == tamanho_alvo) {
        processa_solucao(solucao_parcial);
        return;
    }

    // 2. TRANSIÇÕES (Iterar pelas opções/caminhos disponíveis a partir daqui)
    // Pode ser um loop (ex: 0 a N-1), iterar sobre vizinhos em um grafo, etc.
    for (int opcao = 0; opcao < /* NUMERO_DE_OPCOES */; opcao++) {
        
        // 3. PODA (Pruning)
        if (!eh_valida(opcao)) {
            continue; // Se não for válido, pula esta opção imediatamente
        }

        // 4. FAZER A ESCOLHA
        solucao_parcial.push_back(opcao);
        // Ex: Marcar vis[opcao] = true;

        // 5. CHAMADA RECURSIVA (Avançar para o próximo passo)
        backtrack(estado + 1, tamanho_alvo, solucao_parcial);

        // 6. DESFAZER A ESCOLHA (Backtrack / "Voltar atrás")
        solucao_parcial.pop_back();
        // Ex: Desmarcar vis[opcao] = false;
    }
