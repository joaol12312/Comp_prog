#define FORi(n) for(int i=0;i<n;i++)
int mapa[20][20];
int dl[4]={0,0,1,-1};
int dc[4]={1,-1,0,0};
bool valida(int b, int d){
    return(b>=1 && b<=19 && d<=19 && d>=1);
}
pair<int ,vector<pair<int,int>>> jogo(int l,int c, int cor){
    vector<pair<int,int>> pedras;
    int liberdade=0;
    bool ja_visitou[20][20]={false};
    queue <pair<int,int>> fila;
    fila.push({l,c});
    ja_visitou[l][c]=true;
    while(!fila.empty()){
        auto[l1,c1]=fila.front();

                    break;
                }else if(mapa[l1+dl[i]][c1+dc[i]]==cor && !ja_visitou[l1+dl[i]][c1+dc[i]]){
                    ja_visitou[l1+dl[i]][c1+dc[i]]=true;
                    fila.push({l1+dl[i],c1+dc[i]});
                }
            }
        }
        if(liberdade)break;
    }
    return {liberdade, pedras};
}
