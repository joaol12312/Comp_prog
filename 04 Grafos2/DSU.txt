23. MODELO: DSU (Disjoint Set Union com Union by Size - STRUCT)
-------------------------------------------------------------------------------
// Estrutura DSU otimizada com "Path Compression" e "Union by Size".
// A união baseada no tamanho (sz) garante que a árvore menor seja anexada
// à raiz da árvore maior, mantendo a altura da estrutura extremamente baixa.
// Complexidade de Tempo: O(α(N)) amortizado | Espaço: O(N)

struct DSU {
    vector<int> pai, sz;
    DSU(int n) : pai(n+1), sz(n+1, 1) {
        for(int i=0; i<=n; i++) pai[i] = i;
    }
    
    int find(int u){
        if (pai[u] == u) return u;
        return pai[u] = find(pai[u]);
    }
    
    void join(int u, int v){
        u = find(u), v = find(v);
        
        if(u == v) return;
        if(sz[v] > sz[u]) swap(u, v);
        
        pai[v] = u;
        sz[u] += sz[v];
    }
};
