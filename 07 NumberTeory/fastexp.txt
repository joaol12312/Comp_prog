45. MODELO: EXPONENCIAÇÃO RÁPIDA MODULAR (Binary Exponentiation)
-------------------------------------------------------------------------------
// Calcula (a^b) % MOD em O(log b) utilizando manipulação de bits.
// Requer base 'a' e expoente 'b' não-negativo (b >= 0).
// Complexidade de Tempo: O(log b) | Espaço: O(1)

long long fexp(long long a, long long b, long long mod) {
    long long ans = 1;
    a %= mod;
    while (b > 0) {
        if (b & 1) ans = (ans * a) % mod; // Multiplica se o bit atual estiver aceso
        a = (a * a) % mod;               // Eleva a base ao quadrado
        b >>= 1;                         // Desloca o expoente 1 bit à direita
    }
    return ans;
}

47. MODELO: EXPONENCIAÇÃO RÁPIDA DE MATRIZES (Matrix Fast Exponentiation)
-------------------------------------------------------------------------------
// Eleva uma matriz quadrada N x N ao expoente 'exp' módulo MOD em O(N^3 * log exp).
// Constrói a Matriz Identidade 'ret' e acumula as potências sucessivas de 'T' (T^1, T^2, T^4...).
// Complexidade de Tempo: O(N^3 * log exp) | Espaço: O(N^2)

typedef vector<vector<long long>> Matrix;

Matrix fexp_matrix(Matrix T, long long exp) {
    int n = T.size();
    Matrix ret(n, vector<long long>(n, 0));
    
    // Transforma a matriz de retorno (ret) na Matriz Identidade (o "1" das matrizes)
    for (int i = 0; i < n; i++) {
        ret[i][i] = 1;
    }

    Matrix pot = T; // Começa como T^1

    // Laço da exponenciação bit a bit
    while (exp > 0) {
        if (exp % 2 == 1) { // Se o bit está aceso, multiplica na resposta final
            ret = multiplica(ret, pot);
        }
        // Dobra a potência da matriz (T^1 vira T^2, T^2 vira T^4...)
        pot = multiplica(pot, pot);
        exp /= 2; // Passa para o próximo bit
    }

    return ret;
}
