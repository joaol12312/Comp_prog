45. MODELO: EXPONENCIAÇÃO RÁPIDA MODULAR (Binary Exponentiation)
-------------------------------------------------------------------------------
// Calcula (a^b) % MOD em O(log b) utilizando manipulação de bits.
// Requer base 'a' e expoente 'b' não-negativo (b >= 0).
// Complexidade de Tempo: O(log b) | Espaço: O(1)

long long fexp(long long a, long long b, long long mod) {
    long long ans = 1;
    a %= mod;
    while (b > 0) {
        if (b & 1) ans = (ans * a) % mod; // Multiplica se o bit atual estiver aceso
        a = (a * a) % mod;               // Eleva a base ao quadrado
        b >>= 1;                         // Desloca o expoente 1 bit à direita
    }
    return ans;
}

47. MODELO: EXPONENCIAÇÃO RÁPIDA DE MATRIZES (Matrix Exponentiation via Vector)
-------------------------------------------------------------------------------
// Eleva uma matriz quadrada N x N ao expoente 'exp' módulo MOD em O(N^3 * log exp).
// Constrói a matriz Identidade dinamicamente com base nas dimensões da matriz base.
// Complexidade de Tempo: O(N^3 * log exp) | Espaço: O(N^2)

// Retorna a matriz Identidade N x N (1s na diagonal principal, 0s no restante)
Matrix matriz_identidade(int n) {
    Matrix I(n, vector<long long>(n, 0));
    for (int i = 0; i < n; i++) {
        I[i][i] = 1;
    }
    return I;
}

// Eleva a matriz 'base' ao expoente 'exp'
Matrix matrix_fexp(Matrix base, long long exp) {
    int n = base.size();
    Matrix ans = matriz_identidade(n);

    while (exp > 0) {
        if (exp & 1) ans = multiplica(ans, base);
        base = multiplica(base, base);
        exp >>= 1;
    }

    return ans;
}
