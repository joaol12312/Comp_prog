46. MODELO: MULTIPLICAÇÃO DE MATRIZES (Matrix Multiplication via Vector)
-------------------------------------------------------------------------------
// Realiza a multiplicação de duas matrizes quadradas N x N utilizando vector.
// Aplica aritmética modular (MOD) em cada operação.
// Complexidade de Tempo: O(N^3) | Espaço: O(N^2)

typedef vector<vector<long long>> Matrix;
const long long MOD = 1e9 + 7;

// Multiplica duas matrizes N x N (onde N = A.size()) módulo MOD
Matrix multiplica(const Matrix& A, const Matrix& B) {
    int n = A.size();
    Matrix C(n, vector<long long>(n, 0));

    for (int i = 0; i < n; i++) {
        for (int k = 0; k < n; k++) {
            for (int j = 0; j < n; j++) {
                long long prod = (A[i][k] * B[k][j]) % MOD;
                C[i][j] = (C[i][j] + prod) % MOD;
            }
        }
    }
    return C;
}
